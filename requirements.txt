# Ecalender 系统依赖清单
# 这些依赖需要从系统包管理器安装，不在项目文件夹内
#
# 安装命令（Ubuntu/WSL）：
#   sudo apt-get install -y $(grep -v "^#" requirements.txt | tr '\n' ' ')

# ============================================================
# 编译工具链
# ============================================================
build-essential          # g++, make 等基础编译工具
cmake                    # 构建系统

# ============================================================
# Qt5 图形界面（GUI 模式必需）
# ============================================================
qtbase5-dev              # Qt5 核心库（QWidgets, QGui 等）
libqt5widgets5           # Qt5 控件库
libqt5network5           # Qt5 网络模块

# ============================================================
# 语音识别（Vosk）
# ============================================================
# 模型文件（约 2GB，需手动下载）：
#   旧：vosk-model-small-cn-0.22（42MB，准确率低）
#   新：vosk-model-cn-0.22（2.0GB，推荐）
# 下载地址：https://alphacephei.com/vosk/models
# 第三方库：third_party/vosk/libvosk.so（已包含在项目中）
libpthread-stubs0t       # 线程支持（系统自带）

# ============================================================
# 音频处理（录音 + 播放 + 格式转换）
# ============================================================
ffmpeg                   # 音频格式转换（WAV 转 16kHz mono，必需）
alsa-utils               # arecord（WSL 录音工具）
pulseaudio-utils         # paplay/parec（音频播放/录制，WSLg 环境）
sox                      # rec（录音备用工具）

# ============================================================
# AI 解析（DeepSeek API）
# ============================================================
curl                     # 调用 DeepSeek API（替代 libcurl）
# API Key 需在 .env 文件中配置：
#   DEEPSEEK_API_KEY=sk-your-key-here

# ============================================================
# 中文输入法（Qt GUI 必需）
# ============================================================
ibus                     # 输入法框架（推荐，WSLg 兼容好）
ibus-libpinyin           # 拼音输入引擎
# 备选：fcitx + fcitx-pinyin

# ============================================================
# 中文字体（显示中文必需）
# ============================================================
fonts-noto-cjk           # Noto Sans CJK 中文字体
fonts-noto-color-emoji   # Emoji 字体（可选）

# ============================================================
# 其他工具
# ============================================================
wget                     # 下载模型文件
unzip                    # 解压（WSL 可能不自带）
